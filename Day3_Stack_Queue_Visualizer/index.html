Stack Implementation (C Language)

#include <stdio.h>
#include <stdbool.h>

#define MAX_SIZE 6

typedef struct {
    int items[MAX_SIZE];
    int top;
} Stack;

void initStack(Stack *s) {
    s->top = -1;
}

bool isFull(Stack *s) {
    return s->top == MAX_SIZE - 1;
}

bool isEmpty(Stack *s) {
    return s->top == -1;
}

void push(Stack *s, int element) {
    if (isFull(s)) {
        printf("Overflow Error\\n");
        return;
    }
    s->items[++(s->top)] = element;
}

int pop(Stack *s) {
    if (isEmpty(s)) {
        printf("Underflow Error\\n");
        return -1;
    }
    return s->items[(s->top)--];
}

int peek(Stack *s) {
    if (isEmpty(s)) return -1;
    return s->items[s->top];
}

Queue Implementation (C Language)

#include <stdio.h>
#include <stdbool.h>

#define MAX_SIZE 6

typedef struct {
    int items[MAX_SIZE];
    int front;
    int rear;
} Queue;

void initQueue(Queue *q) {
    q->front = -1;
    q->rear = -1;
}

bool isFull(Queue *q) {
    return q->rear == MAX_SIZE - 1;
}

bool isEmpty(Queue *q) {
    return q->front == -1 || q->front > q->rear;
}

void enqueue(Queue *q, int element) {
    if (isFull(q)) {
        printf("Overflow Error\\n");
        return;
    }
    if (q->front == -1) q->front = 0;
    q->items[++(q->rear)] = element;
}

int dequeue(Queue *q) {
    if (isEmpty(q)) {
        printf("Underflow Error\\n");
        return -1;
    }
    return q->items[(q->front)++];
}

int peek(Queue *q) {
    if (isEmpty(q)) return -1;
    return q->items[q->front];
}

Stack vs. Queue Comparison

Feature Stack 🥞 Queue 🚶‍♂️
Principle LIFO (Last-In, First-Out) FIFO (First-In, First-Out)
Insertion Push() - Added to the Top Enqueue() - Added to the Rear
Deletion Pop() - Removed from the Top Dequeue() - Removed from the Front
Time Complexity O(1) for Push/Pop O(1) for Enqueue/Dequeue
Use Cases Undo mechanisms, Browser History, Function Call Stack (Recursion) Task scheduling, Printer jobs, Message Queues (Event Loop)
Real World Analogy A stack of plates in a cafeteria. You only take from the top. A line of people waiting for tickets. First person in line goes first.
